cmake_minimum_required(VERSION 3.24)
project(Shaper VERSION 0.8.2 LANGUAGES C CXX)
add_compile_options("$<$<COMPILE_LANG_AND_ID:CXX,MSVC>:/utf-8>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/scripts/prepare_sources.py" "${CMAKE_BINARY_DIR}/modules" COMMAND_ERROR_IS_FATAL ANY)
add_subdirectory(upstream/gate12/libs/JUCE)
set(shaper_formats VST3 Standalone)
if(APPLE)
    list(APPEND shaper_formats AU)
endif()
juce_add_plugin(Shaper COMPANY_NAME "FAZIL IRAK" BUNDLE_ID "org.shaper.suite"
    PLUGIN_MANUFACTURER_CODE Shpr PLUGIN_CODE Sh12 PRODUCT_NAME "Shaper"
    FORMATS ${shaper_formats} IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
    MICROPHONE_PERMISSION_ENABLED TRUE
    MICROPHONE_PERMISSION_TEXT "Shaper uses audio input to process sound."
    COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(Shaper)
# This JUCE revision omits Info.txt from its Windows resource dependencies.
# Regenerate the executable's version/company metadata after a version bump.
if(WIN32)
    get_target_property(shaper_generated Shaper JUCE_GENERATED_SOURCES_DIRECTORY)
    get_target_property(shaper_info Shaper JUCE_INFO_FILE)
    add_custom_command(OUTPUT "${shaper_generated}/Shaper_resources.rc" APPEND DEPENDS "${shaper_info}")
endif()
target_compile_definitions(Shaper PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(Shaper PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
target_sources(Shaper PRIVATE Source/Processor.cpp Source/Editor.cpp Source/EffectPalette.cpp Source/AboutPanel.cpp Source/MultibandProcessor.cpp Source/FactoryPresets.cpp Source/EnvelopeEffect.cpp Source/EnvelopeEditor.cpp)
target_include_directories(Shaper PRIVATE Source)
foreach(module gate12 time12 filtr reevr pan12 volumef satcurve pitchcurve)
    file(GLOB_RECURSE module_sources CONFIGURE_DEPENDS "${CMAKE_BINARY_DIR}/modules/${module}/*.cpp")
    if(module STREQUAL "pan12" OR module STREQUAL "volumef" OR module STREQUAL "satcurve" OR module STREQUAL "pitchcurve")
        set(resource_target gate12_resources)
    else()
        file(GLOB_RECURSE resources CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/upstream/${module}/resources/*.*")
        if(module STREQUAL "reevr")
            # Fazıl Irak: dağıtım izni doğrulanmamış IR yerine özgün sentetik kaynak.
            list(FILTER resources EXCLUDE REGEX "/impulses/Hall Quad\\.flac$")
            list(APPEND resources "${CMAKE_SOURCE_DIR}/Resources/impulses/Shaper_Hall_Quad.wav")
        endif()
        juce_add_binary_data(${module}_resources NAMESPACE ${module}_data HEADER_NAME ${module}_data.h SOURCES ${resources})
        set(resource_target ${module}_resources)
    endif()
    add_library(${module}_embedded STATIC ${module_sources})
    target_include_directories(${module}_embedded PRIVATE "${CMAKE_SOURCE_DIR}/Source/Embedded" "${CMAKE_BINARY_DIR}/modules/${module}" "${CMAKE_SOURCE_DIR}/upstream/reevr/libs/FFTConvolver")
    # Compile JUCE itself once in the outer plugin. Embedded modules only need its headers.
    target_link_libraries(${module}_embedded PRIVATE ${resource_target})
    target_include_directories(${module}_embedded PRIVATE "${CMAKE_SOURCE_DIR}/upstream/gate12/libs/JUCE/modules")
    target_compile_definitions(${module}_embedded PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
    foreach(juce_module audio_basics audio_devices audio_formats audio_processors audio_utils core data_structures dsp events graphics gui_basics gui_extra)
        target_compile_definitions(${module}_embedded PRIVATE JUCE_MODULE_AVAILABLE_juce_${juce_module}=1)
    endforeach()
    target_compile_definitions(${module}_embedded PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0
        JucePlugin_Name="${module}" JucePlugin_WantsMidiInput=1 JucePlugin_ProducesMidiOutput=1 JucePlugin_IsMidiEffect=0 JucePlugin_IsSynth=0 PROJECT_VERSION="0.6.0")
    target_link_libraries(Shaper PRIVATE ${module}_embedded ${resource_target})
endforeach()
add_library(irak_embedded STATIC Source/Saturation/SaturationDSP.cpp Source/Saturation/PluginProcessor.cpp)
target_include_directories(irak_embedded PRIVATE Source/Embedded upstream/gate12/libs/JUCE/modules PUBLIC Source/Saturation)
target_compile_definitions(irak_embedded PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(Shaper PRIVATE irak_embedded)
add_library(pitch_embedded STATIC Source/Pitch/PitchProcessor.cpp)
target_include_directories(pitch_embedded PRIVATE Source upstream/gate12/libs/JUCE/modules upstream/signalsmith-stretch upstream/signalsmith-linear/include)
target_compile_definitions(pitch_embedded PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_compile_definitions(irak_embedded PRIVATE SHAPER_VERSION_STRING="${PROJECT_VERSION}")
target_compile_definitions(pitch_embedded PRIVATE SHAPER_VERSION_STRING="${PROJECT_VERSION}")
target_link_libraries(Shaper PRIVATE pitch_embedded)
file(GLOB fft_sources "${CMAKE_SOURCE_DIR}/upstream/reevr/libs/FFTConvolver/*.cpp")
target_sources(reevr_embedded PRIVATE ${fft_sources})
